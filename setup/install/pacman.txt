# shell + terminal
zsh

# browser

# audio: PipeWire stack (server in /usr/bin/pipewire from the hyprland deps;
# wireplumber is the session manager, pipewire-pulse/alsa are API shims for
# apps that talk to PulseAudio/ALSA respectively).
wireplumber
pipewire-pulse
pipewire-alsa

# audio: Apple silicon speakers specifically. The kernel driver mutes the
# speakers until speakersafetyd is running (DSP safety filter); asahi-audio
# provides the per-model PipeWire/Wireplumber profiles.
asahi-audio
speakersafetyd

# hardware controls: read/write backlight devices (screen + keyboard).
# Used by the F1/F2 binds in hyprland.conf via the screen-brightness script.
brightnessctl

# media playback control via MPRIS (used by F7/F8/F9 binds in hyprland.conf
# through swayosd-client's --playerctl wrapper).
playerctl

# OSD: on-screen pill that pops up when volume / brightness changes. Purely
# visual — swap for `wob`, `avizo`, or remove entirely without touching audio.
swayosd

# Emoji picker (Ctrl+Cmd+Space binding in hyprland.conf, macOS-style).
# rofimoji is the picker; fuzzel is the fast minimal wlroots selector it
# draws into; wtype is the wayland typing tool that injects the chosen
# emoji into the focused window.
rofimoji
fuzzel
wtype

# Tiny Wayland terminal we don't actually use — present only because
# hyprland-welcome's "is there a terminal?" check hardcodes a list of
# specific terminal binaries (kitty/alacritty/foot/...) and ghostty
# isn't on it. foot is the lightest of the supported ones.
foot

# Hyprland compositor. Other hypr* packages (hyprutils, hyprlang,
# hyprcursor, hyprgraphics, hyprtoolkit, hyprwayland-scanner, hyprwire,
# hyprland-guiutils) are pulled transitively. aquamarine is NOT here —
# install-aquamarine.sh builds a patched version before this step runs.
hyprland

# Hyprland-specific xdg-desktop-portal backend (Screencast / Screenshot
# for browser tab sharing, Discord screen share, screenshot tools).
xdg-desktop-portal-hyprland

# Qt theming engine + Qt6 settings. The xdg-desktop-portal-hyprland share
# picker and any future Qt apps inherit a Catppuccin look via Kvantum
# instead of the default unstyled gray. catppuccin-kvantum theme is
# pulled from the AUR.
kvantum
qt6ct

# Notification daemon. Implements org.freedesktop.Notifications so apps
# (chromium, electron, libnotify-using CLIs) actually display toasts.
# Without one, those apps can freeze waiting for the D-Bus interface.
mako

# Client side of the notification protocol — provides `notify-send` for
# shell scripts and is what most CLI/GUI apps link against to emit toasts.
libnotify

# JSON parser. Used by fuzzel-polkit-agent (dotfiles/local/bin/) to read
# auth requests piped from cmd-polkit-agent.
jq

# Clipboard history. wl-clipboard provides wl-paste/wl-copy; cliphist is
# the daemon that records every copy event and lets you paste older ones.
# Wired to Cmd+Shift+V via a fuzzel picker in hyprland.conf.
cliphist
wl-clipboard

# Top status bar. Empty by default; modules added later in
# dotfiles/config/waybar/config.jsonc.
waybar

# TUI file manager (Cmd+E in hyprland.conf). imagemagick + ffmpegthumbnailer
# extend yazi's preview support to extra image formats and video thumbnails.
yazi
imagemagick
ffmpegthumbnailer

# Wallpaper daemon. Invoked from hyprland.conf with `-i <image> -m fill`;
# no config file. Switched off hyprpaper because hyprpaper 0.8.x crashes on
# Asahi (empty EDID/monitor description → null std::string deref in the
# hyprtoolkit rewrite). swaybg is the Hyprland wiki's recommended simple
# alternative and just calls into wlr-layer-shell, no monitor introspection.
swaybg

# AUR packages installed via yay (one per line; '#' comments allowed).

# Catppuccin GTK themes. Drive the appearance of GTK apps —
# chromium file picker (via xdg-desktop-portal-gtk) is the main user.
# Accent variant is chosen via the `gtk-theme-name` file in each
# orchard-theme dir and applied by set-theme on swap. Install one
# package per Catppuccin flavor that has an orchard-themes entry.
catppuccin-gtk-theme-frappe
catppuccin-gtk-theme-macchiato

# Everforest GTK theme: see setup/install/install-everforest-gtk.sh
# instead. The AUR package (everforest-gtk-theme-git) depends on
# gtk-engine-murrine, whose autotools `configure` can't auto-detect
# aarch64 and fails to build on Asahi. murrine is a GTK2-only runtime
# engine; chromium's file picker uses GTK3 so we don't need it. We
# install Fausto's theme directly from upstream instead, which only
# needs sassc.

# Backend for fuzzel-polkit-agent (vendored at dotfiles/local/bin/).
# Provides `cmd-polkit-agent`, which registers with polkit and pipes
# JSON to/from a chosen UI command — in our case, fuzzel.
cmd-polkit-git

# Catppuccin Kvantum theme. Selected variant set in
# dotfiles/config/Kvantum/kvantum.kvconfig.
kvantum-theme-catppuccin-git

# Apple's font collection — SF Pro for system UI, SF Mono as an
# alternative monospace, plus New York serif. Picked over the targeted
# `otf-apple-sf-pro` AUR package because that one's checksums lag every
# time Apple silently re-publishes the DMG; this metapackage is more
# actively maintained.
apple-fonts

# Phosphor Icons — single-author icon font with consistent stroke
# weights and baselines, used for waybar's right-side status icons
# (battery, wifi, bell, shield, etc.). Replaces the previous mix of
# nerd-font glyphs that had visibly inconsistent weights.
ttf-phosphor-icons

# BlueBubbles desktop client — pairs with BlueBubbles Server running on
# an always-on Mac (the Mac handles iMessage; this is the Linux UI).
bluebubbles-bin

# NordVPN official client. CLI-driven (`nordvpn login`, `nordvpn c`).
# Daemon enable, group membership, and DNS override are wired up by
# configure-nordvpn.sh. `nordvpn login` / `connect` stay manual.
nordvpn-bin
# gcalcli — Google Calendar CLI. Talks to the REST API (not CalDAV)
# so it sidesteps the URL-encoded UID issues vdirsyncer choked on.
# First run: `gcalcli init` walks through OAuth via browser. After
# that: `gcalcli calw` (week view), `gcalcli agenda` (upcoming list).
gcalcli
